            }

class LocationTrackingService : Service() {

    companion object {
        private const val TAG = "CallRadar"
        private const val CHANNEL_ID = "callradar_location"
        private const val NOTIF_ID = 3

        // ?꾩옱 ?꾩튂 (?꾩뿭 怨듭쑀)
        var currentLat: Double = 0.0
        var currentLng: Double = 0.0
        var isMoving: Boolean = false
        var currentSpeed: Float = 0f
    }

    private lateinit var fusedLocationClient: FusedLocationProviderClient
    private lateinit var locationCallback: LocationCallback
    private var isTracking = false

    // ?댄뻾 媛먯???
    private var lastLat: Double = 0.0
    private var lastLng: Double = 0.0
    private var stationaryCount: Int = 0
    private val STATIONARY_THRESHOLD = 6 // 30珥??댁긽 ?뺤? = ?湲?以?

    override fun onCreate() {
        super.onCreate()
        fusedLocationClient = LocationServices.getFusedLocationProviderClient(this)
        setupLocationCallback()
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        startForeground(NOTIF_ID, createNotification())
        startLocationTracking()
        return START_STICKY
    }

    private fun setupLocationCallback() {
        locationCallback = object : LocationCallback() {
            override fun onLocationResult(result: LocationResult) {
                val location = result.lastLocation ?: return
                updateLocation(location)
            }
        }
    }

    private fun startLocationTracking() {
        if (isTracking) return

        val request = LocationRequest.Builder(
            Priority.PRIORITY_HIGH_ACCURACY, 5000L // 5珥덈쭏??
        ).apply {
            setMinUpdateDistanceMeters(10f) // 10m ?댁긽 ?대룞 ??
            setGranularity(Granularity.GRANULARITY_FINE)
        }.build()

        try {
            fusedLocationClient.requestLocationUpdates(
                request, locationCallback, Looper.getMainLooper()
            )
            isTracking = true
            Log.d(TAG, "??GPS 異붿쟻 ?쒖옉")
        } catch (e: SecurityException) {
            Log.e(TAG, "?꾩튂 沅뚰븳 ?놁쓬: ${e.message}")
        }
    }

    private fun updateLocation(location: Location) {
        currentLat = location.latitude
        currentLng = location.longitude
        currentSpeed = location.speed * 3.6f // m/s ??km/h

        // NaviIntentReceiver???꾩튂 ?꾨떖
        // (?ㅼ젣 ?깆뿉?쒕뒗 諛붿씤???먮뒗 釉뚮줈?쒖틦?ㅽ듃濡??꾨떖)
        isMoving = currentSpeed > 5f // 5km/h ?댁긽?대㈃ ?대룞 以?

        // ?뺤?/?대룞 媛먯?
        val distance = FloatArray(1)
        Location.distanceBetween(lastLat, lastLng, currentLat, currentLng, distance)

        if (distance[0] < 20f) { // 20m 誘몃쭔 ?대룞
            stationaryCount++
        } else {
            stationaryCount = 0
            lastLat = currentLat
            lastLng = currentLng
        }

        val status = when {
            currentSpeed > 30 -> "怨좎냽?대룞"
            currentSpeed > 5 -> "?대룞以?
            stationaryCount > STATIONARY_THRESHOLD -> "?湲곗쨷"
            else -> "?쒗뻾"
        }

        Log.d(TAG, "?뱧 ?꾩튂: $currentLat, $currentLng | ?띾룄: ${currentSpeed.toInt()}km/h | $status")
    }

    private fun createNotification(): Notification {
        val channel = NotificationChannel(
            CHANNEL_ID, "肄쒕젅?대뜑 ?꾩튂 異붿쟻",
            NotificationManager.IMPORTANCE_LOW
        )
        getSystemService(NotificationManager::class.java).createNotificationChannel(channel)

        return NotificationCompat.Builder(this, CHANNEL_ID)
            .setContentTitle("肄쒕젅?대뜑 ?댄뻾 以?)
            .setContentText("GPS 異붿쟻 以?..")
            .setSmallIcon(android.R.drawable.ic_menu_mylocation)
            .setOngoing(true)
            .build()
    }

    override fun onDestroy() {
        fusedLocationClient.removeLocationUpdates(locationCallback)
        isTracking = false
        Log.d(TAG, "GPS 異붿쟻 醫낅즺")
        super.onDestroy()
    }

    @Composable
    fun AppContent() {
        val prefs = getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE)
        var isLoggedIn by remember { mutableStateOf(prefs.getString(KEY_NICKNAME, null) != null) }
        var userNickname by remember { mutableStateOf(prefs.getString(KEY_NICKNAME, "") ?: "") }
        var userId by remember { mutableStateOf(prefs.getString(KEY_USER_ID, "") ?: "") }
        var onboardingDone by remember { mutableStateOf(prefs.getBoolean(KEY_ONBOARDING_DONE, false)) }
        when {
            !isLoggedIn -> LoginScreen(onLoginSuccess = { uid, nickname ->
                prefs.edit().putString(KEY_USER_ID, uid).putString(KEY_NICKNAME, nickname).apply()
                userId = uid; userNickname = nickname; isLoggedIn = true
            })
            !onboardingDone -> OnboardingScreen(nickname = userNickname, onDone = {
                prefs.edit().putBoolean(KEY_ONBOARDING_DONE, true).apply(); onboardingDone = true
            })
            else -> MainWithTabs(nickname = userNickname, userId = userId, onEndShift = {
                stopService(Intent(this, LocationTrackingService::class.java))
            }, onLogout = {
                prefs.edit().clear().apply()
                stopService(Intent(this, LocationTrackingService::class.java))
                isLoggedIn = false; userNickname = ""; userId = ""; onboardingDone = false
            })
        }
    }

@Composable
fun CallRadarTheme(
    darkTheme: Boolean = isSystemInDarkTheme(),
    // Dynamic color is available on Android 12+
    dynamicColor: Boolean = true,
    content: @Composable () -> Unit
) {
    val colorScheme = when {
        dynamicColor && Build.VERSION.SDK_INT >= Build.VERSION_CODES.S -> {
            val context = LocalContext.current
            if (darkTheme) dynamicDarkColorScheme(context) else dynamicLightColorScheme(context)
        }

        darkTheme -> DarkColorScheme
        else -> LightColorScheme
    }

    MaterialTheme(
        colorScheme = colorScheme,
        typography = Typography,
        content = content
    )
